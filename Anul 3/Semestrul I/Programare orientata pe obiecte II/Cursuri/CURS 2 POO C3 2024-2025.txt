import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class Main {

	public static void main(String[] args) {
		final int N_MAX_LINII=1000;
		String biblioteca[]=new String[N_MAX_LINII];
		int N=0;//numarul real de linii din fisier
		try {
			FileReader fr=new FileReader("carti.txt");
			BufferedReader bfr=new BufferedReader(fr);
			//citire linie cu linie:
			int i=0;//index in vector biblioteca[]
			for(;;) {
				String s=bfr.readLine();
				if(s==null)break;
				biblioteca[i]=s;
				i++;
			}
			N=i;
			bfr.close();
			fr.close();
		}catch(IOException e) {
			System.exit(1);
		}
		//Verificam ca a fost salvat corect fisieru in vector:
		for(int i=0;i<N;i++)
			System.out.println(biblioteca[i]);
	}

}
