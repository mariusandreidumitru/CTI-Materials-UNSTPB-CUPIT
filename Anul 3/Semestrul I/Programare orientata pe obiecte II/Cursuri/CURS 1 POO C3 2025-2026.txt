import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		int N;
		Scanner sc=new Scanner(System.in);
		System.out.print("N=");
		N=sc.nextInt();
		int a[]=new int[N];
		for(int i=0;i<N;i++) {
			System.out.print("nr=");
			a[i]=sc.nextInt();
		}
		int maxim=maxVector(a);
		System.out.println("max="+maxim);
	}
	static int maxVector(int a[]) {
		int max=a[0];
		for(int i=1;i<a.length;i++)
			if(a[i]>max)max=a[i];
		return max;
	}	
}

B. Maximul dintr-un ArrayList de numere intregi
ArrayList<Integer>
Citim nr intregi de la tastatura pana tastez 0, intr-un ArrayList
import java.util.ArrayList;
import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		ArrayList<Integer> lista=new ArrayList();
		for(;;) {
			System.out.print("nr=");
			int nr=sc.nextInt();
			if(nr==0)break;
			lista.add(nr);
		}
		int max=lista.get(0);
		//parcurg lista:
		for(int i=1;i<lista.size();i++)
			if(lista.get(i)>max)
				max=lista.get(i);
		System.out.println("maxim="+max);
	}

}

max=77

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class Main {

	public static void main(String[] args) {
		try {
			FileReader fr=new FileReader("date.txt");
			BufferedReader bfr=new BufferedReader(fr);
			//pres. ca exista cel putin un numar in fisier:
			String s=bfr.readLine();
			int max=Integer.parseInt(s);//"123"--->123
			//citim restul numerelor
			for(;;) {
				s=bfr.readLine();
				if(s==null)break;
				int nr=Integer.parseInt(s);
				if(nr>max)max=nr;
			}
			bfr.close();fr.close();
			System.out.println("maxim="+max);
	    }catch(IOException e) {
	    	System.out.println("eroare fisier");
	    	System.exit(0);
	    }
	}
}
