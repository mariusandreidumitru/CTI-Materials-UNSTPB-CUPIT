							pw.println(nr);

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.net.Socket;

public class FirClient extends Thread {
	private Socket cs;
	private BufferedReader bfr;
	private PrintWriter pw;
	public FirClient(Socket cs) {
		try {
			this.cs=cs;
			pw=new PrintWriter(cs.getOutputStream());
			bfr=new BufferedReader(new InputStreamReader(cs.getInputStream()));
		}catch(Exception e) {
		 e.printStackTrace();
				System.exit(0);
		}
	}
	public void run() {
		try {
			for(;;) {
				String textln = bfr.readLine();
				if(textln.equals(""))break;
				int N=Integer.parseInt(textln);
				//primul numar din cele N trimise, voi initializa maximul cu el:
				int max=Integer.parseInt(bfr.readLine());
				//Citesc de la clent restul de N-1 numere:
				for(int i=1;i<=N-1;i++) {
					int numar=Integer.parseInt(bfr.readLine());
					if(numar>max)max=numar;
				}
				pw.println("Maximul este: "+max);
				pw.flush();
			}
			}catch(Exception e) {
				e.printStackTrace();
				System.exit(0);
		}
	}
}

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.net.Socket;

public class FirClient extends Thread {
	private Socket cs;
	private BufferedReader bfr;
	private PrintWriter pw;
	public FirClient(Socket cs) {
		try {
			this.cs=cs;
			pw=new PrintWriter(cs.getOutputStream());
			bfr=new BufferedReader(new InputStreamReader(cs.getInputStream()));
		}catch(Exception e) {
		 e.printStackTrace();
				System.exit(0);
		}
	}
	public void run() {
		try {
			for(;;) {
				String toateNumerele = bfr.readLine();
				if(toateNumerele.equals(""))break;//s-a terminat clientul
				//Despachetam numerele:
				String s[]=toateNumerele.split(" ");
				//numarul de numere, este dimensiunea vectorului s:
				int N=s.length;
				//primul numar din cele N trimise, s[0], voi initializa maximul cu el:
				int max=Integer.parseInt(s[0]);
				//Restul de N-1 numere:
				for(int i=1;i<=N-1;i++) {
					int numar=Integer.parseInt(s[i]);
					if(numar>max)max=numar;
				}
				pw.println("Maximul este: "+max);
				pw.flush();
			}
			}catch(Exception e) {
				e.printStackTrace();
				System.exit(0);
		}
	}
}

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.net.Socket;

public class FirClient extends Thread {
	private Socket cs;
	private BufferedReader bfr;
	private PrintWriter pw;
	public FirClient(Socket cs) {
		try {
			this.cs=cs;
			pw=new PrintWriter(cs.getOutputStream());
			bfr=new BufferedReader(new InputStreamReader(cs.getInputStream()));
		}catch(Exception e) {
		 e.printStackTrace();
				System.exit(0);
		}
	}
	public void run() {
		try {
			for(;;) {
				String sN = bfr.readLine();
				if(sN.equals(""))break;//s-a terminat clientul
				int N=Integer.parseInt(sN);
				//deschid fisierul text:
				FileReader fr=new FileReader("date.txt");
				BufferedReader bf=new BufferedReader(fr);
				int nrLinie=0;
				boolean gasit=false;
				String linieCrt="";
				//Citesc liniile din fisier pana ajungem la linia N:
				for(;;) {
					linieCrt=bf.readLine();
					if(linieCrt==null)break;//s-a terminat fisierul
					nrLinie++;
					if(nrLinie==N) {
						gasit=true;//este linieCrt ce o vom servi clientului
						break;
					}
				}//for;;
				bf.close(); fr.close();
				if(gasit==true) {
				   pw.println("Linia "+N+" este: "+linieCrt);
				   pw.flush();}
				else {
					pw.println("NU EXISTA!");
					pw.flush();
				}
			 }//for;;	
			}catch(Exception e) {
				e.printStackTrace();
				System.exit(0);
		    }
	}//run
}
